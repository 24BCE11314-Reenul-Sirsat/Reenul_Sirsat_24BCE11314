db.products.find(
    {
        $or: [
            { color: { $size: 1 } },
            { color: { $size: 2 } }
        ]
    },
    { _id: 0, name: 1, color: 1 }
)

// Contains both black and white
db.products.find(
    { color: { $all: ["black","white"] } },
    { _id: 0, name: 1, color: 1 }

    {
        $and: [
            { color: "black" },
