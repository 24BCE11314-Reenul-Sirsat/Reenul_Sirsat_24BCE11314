/* Q1. Display employee name with designation and department */

SELECT emp.name AS employee_name,
       job.job_title,
       dept.dept_name
FROM Jobs job
INNER JOIN Employees emp
        ON job.job_id = emp.job_id
INNER JOIN Departments dept
        ON dept.dept_id = emp.dept_id;


/* Q2. Employees earning more than the average salary of IT department */

SELECT name,
       salary
FROM Employees
WHERE salary >
(
    SELECT AVG(salary)
    FROM Employees
    WHERE dept_id =
    (
        SELECT dept_id
        FROM Departments
        WHERE dept_name = 'IT'
    )
);


/* Q3. Total salary spent by every department */

SELECT dept.dept_name,
       SUM(emp.salary) AS salary_expense
FROM Departments dept
JOIN Employees emp
     ON dept.dept_id = emp.dept_id
GROUP BY dept.dept_name;


/* Q4. Departments having more than 2 employees and average salary above 70000 */

SELECT dept.dept_name,
       COUNT(*) AS employee_count,
       AVG(emp.salary) AS average_salary
FROM Departments dept
INNER JOIN Employees emp
        ON dept.dept_id = emp.dept_id
GROUP BY dept.dept_name
HAVING employee_count > 2
   AND average_salary > 70000;


/* Q5. Employee names along with their managers */

SELECT emp.name AS employee,
       mgr.name AS manager
FROM Employees emp
LEFT OUTER JOIN Employees mgr
     ON emp.manager_id = mgr.emp_id;


/* Q6. Three highest paid employees */

SELECT name,
       salary
FROM Employees
ORDER BY salary DESC
LIMIT 3;


/* Q7. Developers located in San Francisco */

SELECT emp.name,
       job.job_title,
       dept.location
FROM Employees emp
JOIN Jobs job
     ON emp.job_id = job.job_id
JOIN Departments dept
     ON emp.dept_id = dept.dept_id
WHERE dept.location = 'San Francisco'
  AND job.job_title = 'Developer';


/* Q8. Number of employees hired year-wise */

SELECT YEAR(hire_date) AS year_joined,
       COUNT(*) AS total_hired
FROM Employees
GROUP BY YEAR(hire_date);


/* Q9. Job roles whose highest salary is below 100000 */

SELECT job.job_title,
       MAX(emp.salary) AS highest_salary
FROM Jobs job
JOIN Employees emp
     ON job.job_id = emp.job_id
GROUP BY job.job_title
HAVING highest_salary < 100000;


/* Q10. Employees working in David's department */

SELECT name
FROM Employees
WHERE dept_id =
(
    SELECT dept_id
    FROM Employees
    WHERE name = 'David'
);


/* Q12. Employees hired from 2018–2020 earning above 60000 */

SELECT name,
       salary,
       hire_date
FROM Employees
WHERE salary > 60000
AND hire_date BETWEEN '2018-01-01' AND '2020-12-31';


/* Q13. Employees earning above department average */

SELECT e.name,
       e.salary,
       e.dept_id
FROM Employees e
WHERE e.salary >
(
    SELECT AVG(x.salary)
    FROM Employees x
    WHERE x.dept_id = e.dept_id
);


/* Q14. Job title and salary of employee ID 504 */

SELECT emp.name,
       jb.job_title,
       emp.salary
FROM Employees emp
INNER JOIN Jobs jb
        ON emp.job_id = jb.job_id
WHERE emp.emp_id = 504;


/* Q15. Minimum salary under each manager */

SELECT manager_id,
       MIN(salary) AS lowest_salary
FROM Employees
WHERE manager_id IS NOT NULL
GROUP BY manager_id;


/* Q16. Departments without Analysts */

SELECT dept_name
FROM Departments
WHERE dept_id NOT IN
(
    SELECT e.dept_id
    FROM Employees e
    INNER JOIN Jobs j
            ON e.job_id = j.job_id
    WHERE j.job_title = 'Analyst'
);


/* Q17. Difference between highest and lowest company salary */

SELECT MAX(salary) - MIN(salary) AS pay_gap
FROM Employees;


/* Q19. Managers supervising more than two employees */

SELECT manager_id,
       COUNT(*) AS team_members
FROM Employees
WHERE manager_id IS NOT NULL
GROUP BY manager_id
HAVING COUNT(*) > 2;


/* Q20. Employee(s) with third highest salary */

SELECT name,
       salary
FROM Employees
WHERE salary =
(
    SELECT DISTINCT salary
    FROM Employees
    ORDER BY salary DESC
    LIMIT 2,1
);


/* Q22. Finance department employees whose names contain 'i' */

SELECT emp.name
FROM Employees emp
INNER JOIN Departments dept
        ON emp.dept_id = dept.dept_id
WHERE dept.dept_name = 'Finance'
  AND emp.name LIKE '%i%';


/* Q23. Total 10% bonus amount generated by each department */

SELECT dept.dept_name,
       SUM(emp.salary * 0.10) AS department_bonus
FROM Employees emp
JOIN Departments dept
     ON emp.dept_id = dept.dept_id
GROUP BY dept.dept_name;


/* Q24. Department with the highest total salary payout */

SELECT dept_name
FROM Departments
WHERE dept_id =
(
    SELECT dept_id
    FROM Employees
    GROUP BY dept_id
    ORDER BY SUM(salary) DESC
    LIMIT 1
);


/* Q25. Employees reporting to Charlie */

SELECT emp.name AS reporting_employee
FROM Employees emp
INNER JOIN Employees manager
        ON emp.manager_id = manager.emp_id
WHERE manager.name = 'Charlie';
