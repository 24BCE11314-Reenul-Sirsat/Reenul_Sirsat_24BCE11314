/* Q1. Display Employee Name, Department Name and Location */

SELECT
    emp.employee_name AS Employee,
    dept.dept_name AS Department,
    dept.location_id AS Location
FROM Departments dept
INNER JOIN Employees emp
        ON dept.dept_id = emp.dept_id;


/* Q2. Project Name, Client Name and Total Hours
   for projects under Consulting department */

SELECT
    pr.project_name,
    pr.client_name,
    SUM(asg.hours_worked) AS Hours_Total
FROM Departments dept
INNER JOIN Projects pr
        ON dept.dept_id = pr.dept_id
INNER JOIN Assignments asg
        ON pr.project_id = asg.project_id
WHERE dept.dept_name = 'Consulting'
GROUP BY pr.project_name, pr.client_name;


/* Q3. Employees who worked on projects for GlobalTech Solutions */

SELECT
    emp.employee_name,
    pr.project_name
FROM Assignments asg
JOIN Employees emp
     ON asg.employee_id = emp.employee_id
JOIN Projects pr
     ON asg.project_id = pr.project_id
WHERE pr.client_name = 'GlobalTech Solutions';


/* Q4. Departments without any employees */

SELECT
    dept.dept_name
FROM Departments dept
LEFT OUTER JOIN Employees emp
             ON dept.dept_id = emp.dept_id
WHERE emp.employee_id IS NULL;


/* Q5. Manager name, department and direct report count */

SELECT
    mgr.employee_name AS Manager_Name,
    dept.dept_name,
    COUNT(emp.employee_id) AS Report_Count
FROM Employees emp
JOIN Employees mgr
     ON emp.manager_id = mgr.employee_id
JOIN Departments dept
     ON mgr.dept_id = dept.dept_id
GROUP BY mgr.employee_name, dept.dept_name;


/* Q6. Average hours worked by each employee */

SELECT
    emp.employee_name,
    ROUND(AVG(asg.hours_worked),2) AS Average_Hours
FROM Employees emp
INNER JOIN Assignments asg
        ON emp.employee_id = asg.employee_id
GROUP BY emp.employee_name;


/* Q7. Departments with average salary
   greater than overall company average */

SELECT
    dept.dept_name
FROM Departments dept
JOIN Employees emp
     ON dept.dept_id = emp.dept_id
GROUP BY dept.dept_name
HAVING AVG(emp.salary) >
(
    SELECT AVG(salary)
    FROM Employees
);


/* Q8. Clients whose projects exceed 500 total hours */

SELECT
    pr.client_name,
    SUM(asg.hours_worked) AS Total_Work_Hours
FROM Projects pr
JOIN Assignments asg
     ON pr.project_id = asg.project_id
GROUP BY pr.client_name
HAVING SUM(asg.hours_worked) > 500;


/* Q10. Projects having more than 5 assigned employees */

SELECT
    pr.project_name,
    COUNT(DISTINCT asg.employee_id) AS Employee_Count
FROM Projects pr
INNER JOIN Assignments asg
        ON pr.project_id = asg.project_id
GROUP BY pr.project_name
HAVING COUNT(DISTINCT asg.employee_id) > 5;


/* Q13. Employees who spent less than 10 hours on a project */

SELECT
    emp.employee_name,
    pr.project_name
FROM Employees emp
JOIN Assignments asg
     ON emp.employee_id = asg.employee_id
JOIN Projects pr
     ON asg.project_id = pr.project_id
WHERE asg.hours_worked < 10;


/* Q14. Projects without any employee assignment */

SELECT
    pr.project_name
FROM Projects pr
LEFT JOIN Assignments asg
       ON pr.project_id = asg.project_id
WHERE asg.assignment_id IS NULL;


/* Q16. Oldest project start date for every employee */

SELECT
    emp.employee_name,
    MIN(asg.start_date) AS First_Project_Date
FROM Employees emp
JOIN Assignments asg
     ON emp.employee_id = asg.employee_id
GROUP BY emp.employee_name;


/* Q17. Projects active during 2023 */

SELECT
    project_name
FROM Projects
WHERE DATE_FORMAT(start_date,'%Y') = '2023';


/* Q18. Show manager, replacing NULL with CEO */

SELECT
    employee_name,
    IFNULL(manager_id,'CEO') AS Reporting_Manager
FROM Employees;


/* Q19. Top 3 departments by total project hours */

SELECT
    dept.dept_name,
    SUM(asg.hours_worked) AS Department_Hours
FROM Departments dept
JOIN Employees emp
     ON dept.dept_id = emp.dept_id
JOIN Assignments asg
     ON emp.employee_id = asg.employee_id
GROUP BY dept.dept_name
ORDER BY Department_Hours DESC
LIMIT 3;


/* Q21. Employee and manager names
   including employees without managers */

SELECT
    emp.employee_name AS Employee_Name,
    mgr.employee_name AS Manager_Name
FROM Employees emp
LEFT OUTER JOIN Employees mgr
               ON emp.manager_id = mgr.employee_id;


/* Q22. CTE showing project-wise unique employee count */

WITH ProjectSummary AS
(
    SELECT
        pr.project_name,
        COUNT(DISTINCT asg.employee_id) AS Employee_Total
    FROM Projects pr
    LEFT JOIN Assignments asg
           ON pr.project_id = asg.project_id
    GROUP BY pr.project_name
)
SELECT
    *
FROM ProjectSummary;
