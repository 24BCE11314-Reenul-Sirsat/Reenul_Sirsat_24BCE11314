/* =========================
   QUERY 1
   Display first names with alias
   ========================= */

SELECT FNAME AS Employee_Name
FROM Employee;


/* =========================
   QUERY 2
   Distinct departments
   ========================= */

SELECT DISTINCT DEPT_NAME
FROM Employee;


/* =========================
   QUERY 3
   Employees named Nitin or Aakash
   ========================= */

SELECT *
FROM Employee
WHERE FNAME IN ('Nitin', 'Aakash');


/* =========================
   QUERY 4
   Names containing letter 'a'
   ========================= */

SELECT *
FROM Employee
WHERE FNAME LIKE '%a%';


/* =========================
   QUERY 5
   Names ending with 'a'
   ========================= */

SELECT *
FROM Employee
WHERE FNAME LIKE '%a';


/* =========================
   QUERY 6
   Admin employees with salary range
   ========================= */

SELECT *
FROM Employee
WHERE DEPT_NAME = 'Admin'
AND EMP_SALARY BETWEEN 100000 AND 500000;


/* =========================
   QUERY 7
   Count employees in Admin
   ========================= */

SELECT COUNT(*) AS Admin_Count
FROM Employee
WHERE DEPT_NAME = 'Admin';


/* =========================
   QUERY 8
   Second highest salary
   ========================= */

SELECT MAX(EMP_SALARY) AS Second_Highest_Salary
FROM Employee
WHERE EMP_SALARY <
(
    SELECT MAX(EMP_SALARY)
    FROM Employee
);


/* =========================
   QUERY 9
   Departments having less than 3 employees
   ========================= */

SELECT DEPT_NAME,
       COUNT(*) AS Employee_Total
FROM Employee
GROUP BY DEPT_NAME
HAVING COUNT(*) < 3;


/* =========================
   QUERY 10
   Employee count department-wise
   ========================= */

SELECT DEPT_NAME,
       COUNT(*) AS Total_Employees
FROM Employee
GROUP BY DEPT_NAME;


/* =========================
   QUERY 11
   Highest paid employee in each department
   ========================= */

SELECT FNAME,
       DEPT_NAME,
       EMP_SALARY
FROM Employee E1
WHERE EMP_SALARY =
(
    SELECT MAX(E2.EMP_SALARY)
    FROM Employee E2
    WHERE E1.DEPT_NAME = E2.DEPT_NAME
);
